function checkTaxDay() { let now = new Date(); let day = now.getDate(); let currentMonthStr = now.getFullYear() + "-" + now.getMonth(); // Xử lý banner Tháng Hoàng Kim if (checkTaxHoliday()) { $('tax-holiday-banner').style.display = 'flex'; let diff = sys.tax.holidayUntil - Date.now(); let d = Math.floor(diff / 86400000); let h = Math.floor((diff % 86400000) / 3600000); let m = Math.floor((diff % 3600000) / 60000); let s = Math.floor((diff % 60000) / 1000); $('tax-holiday-countdown').innerText = `${d} NGÀY ${h.toString().padStart(2,'0')}:${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`; $('tax-banner').style.display = 'none'; return; } else { $('tax-holiday-banner').style.display = 'none'; } // Ngày mùng 5 - Treo cờ đòi nợ if (day === 5 && sys.tax.owed > 0) { $('tax-banner').style.display = 'flex'; $('tax-amount-btn').innerText = `$${Math.floor(sys.tax.owed).toLocaleString()}`; sys.tax.isDueMonth = currentMonthStr; // Đánh dấu tháng này ĐÃ ĐÒI saveAllData(); } else { $('tax-banner').style.display = 'none'; } // Trốn thuế: Qua ngày 5 mà chưa nộp CỦA THÁNG ĐÃ BỊ ĐÒI if (day > 5 && sys.tax.owed > 0 && sys.tax.isDueMonth === currentMonthStr) { let penalty = sys.tax.owed * 2; if (sys.coins >= penalty) { sys.coins -= penalty; } else { sys.coins = 0; // Tịch thu toàn bộ nếu không đủ trả nợ x2 } sys.tax.owed = 0; sys.tax.isDueMonth = ""; // Xóa án tích sau khi phạt saveAllData(); alert(`💀 ÁN PHẠT TRỐN THUẾ!\n\nĐã qua ngày mùng 5 mà bạn chưa nộp thuế! Hệ thống đã cưỡng chế tịch thu $${Math.floor(penalty).toLocaleString()} (x2 Sổ Nợ) từ tài khoản!`); } }
